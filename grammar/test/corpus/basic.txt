public OnPlayerConnect(playerid)
{
    new name[MAX_PLAYER_NAME];
    GetPlayerName(playerid, name, sizeof(name));
    if (playerid >= 0 && playerid < MAX_PLAYERS)
    {
        positions[playerid][0] = 0.0;
    }
    return 1;
}

{
    PLAYER_ID,
    Float:PLAYER_HEALTH = 100
